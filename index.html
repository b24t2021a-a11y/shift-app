/** * シフト自動算出＆カレンダーHTML生成ロジック */ function generateShiftReport(state) { const daysOfWeek = ["日", "月", "火", "水", "木", "金", "土"]; let year = state.year; let month = state.month; let firstDay = new Date(year, month - 1, 1).getDay(); let daysInMonth = new Date(year, month, 0).getDate(); let dailyShifts = {}; let jobStats = {}; for (let j of state.jobs) { jobStats[j.name] = { minutes: 0, salary: 0 }; } for (let d = 1; d <= daysInMonth; d++) { let dateObj = new Date(year, month - 1, d); let dayOfWeek = dateObj.getDay(); let offInfo = state.offDays[d]; if (offInfo && offInfo.type === "all") { dailyShifts[d] = { isWork: false, offReason: offInfo.name || "休み", totalSalary: 0, shifts: [], classCount: 0, eventCount: 0 }; continue; } let openMin = 360; // 06:00 let closeMin = 1380; // 23:00 let timeLine = new Array(closeMin - openMin).fill(true); let classCount = 0; let eventCount = 0; // 授業＋前後1時間除外 if (dayOfWeek >= 1 && dayOfWeek <= 5) { for (let p = 1; p <= state.periodTimes.length; p++) { if (state.timetable.includes(`${dayOfWeek}_${p}`)) { classCount++; let pTime = state.periodTimes[p - 1]; let sMin = timeToMin(pTime.start); let eMin = timeToMin(pTime.end); let ngStart = Math.max(openMin, sMin - 60); let ngEnd = Math.min(closeMin, eMin + 60); for (let m = ngStart; m < ngEnd; m++) timeLine[m - openMin] = false; } } } // 個人予定＋前後1時間除外 if (offInfo && offInfo.intervals) { for (let inv of offInfo.intervals) { eventCount++; let sMin = timeToMin(inv.start); let eMin = timeToMin(inv.end); let ngStart = Math.max(openMin, sMin - 60); let ngEnd = Math.min(closeMin, eMin + 60); for (let m = ngStart; m < ngEnd; m++) timeLine[m - openMin] = false; } } // 空きブロック抽出 (2時間未満除外) let freeBlocks = []; let currStart = null; for (let m = 0; m <= timeLine.length; m++) { if (m < timeLine.length && timeLine[m]) { if (currStart === null) currStart = m; } else { if (currStart !== null) { let s = ceil15(openMin + currStart); let e = floor15(openMin + m); if (e - s >= 120) freeBlocks.push({ start: s, end: e }); currStart = null; } } } let dayShifts = []; let daySalary = 0; let usedJobsInDay = new Set(); for (let block of freeBlocks) { if (!state.allowMultiplePerDay && usedJobsInDay.size > 0 && state.shiftMode === "ai_optimize") { break; } let bestJob = null; let maxScore = -99999; let fitStart = 0; let fitEnd = 0; for (let job of state.jobs) { if (usedJobsInDay.has(job.name)) continue; let jOpen = timeToMin(job.openTime); let jClose = timeToMin(job.closeTime); let s = Math.max(block.start, jOpen); let e = Math.min(block.end, jClose); if (e - s >= 120) { let prefScore = 0; if (state.shiftMode === "ai_optimize") { if (state.timePreference === "morning") { if (s >= 720) prefScore -= 5000; } else if (state.timePreference === "night") { if (s < 720) prefScore -= 10000; } } let durationScore = (e - s); let totalScore = durationScore + prefScore; if (totalScore > maxScore) { if (e - s > 540) e = s + 540; maxScore = totalScore; bestJob = job; fitStart = s; fitEnd = e; } } } if (bestJob && state.shiftMode === "ai_optimize" && state.timePreference === "night" && fitStart < 720) { continue; } if (bestJob && (fitEnd - fitStart >= 120)) { usedJobsInDay.add(bestJob.name); let totalBoundMin = fitEnd - fitStart; let breakMin = totalBoundMin > 330 ? 60 : 0; let actualWorkMin = totalBoundMin - breakMin; let workRatio = actualWorkMin / totalBoundMin; let sSalary = 0; for (let m = fitStart; m < fitEnd; m += 15) { let isPremium = (m >= 360 && m < 540) || (m >= 1320 && m < 1380); let rate = isPremium ? bestJob.wage * 1.25 : bestJob.wage; sSalary += (rate * 0.25) * workRatio; } daySalary += sSalary; dayShifts.push({ jobName: bestJob.name, startStr: minToTime(fitStart), endStr: minToTime(fitEnd), boundDur: totalBoundMin, actualWorkMin: actualWorkMin, breakMin: breakMin, shiftSalary: sSalary, isSuggestRemove: false }); if (jobStats[bestJob.name]) { jobStats[bestJob.name].minutes += actualWorkMin; jobStats[bestJob.name].salary += sSalary; } } } dailyShifts[d] = { isWork: dayShifts.length > 0, totalSalary: daySalary, shifts: dayShifts, intervals: (offInfo && offInfo.intervals) ? offInfo.intervals : [], classCount: classCount, eventCount: eventCount }; } // 6連勤制限チェック let consecutive = 0; for (let d = 1; d <= daysInMonth; d++) { if (dailyShifts[d].isWork) { consecutive++; if (consecutive >= 7) { for (let targetD = d; targetD >= d - 6; targetD--) { let dow = new Date(year, month - 1, targetD).getDay(); if (dow >= 1 && dow <= 5 && dailyShifts[targetD].isWork) { for (let s of dailyShifts[targetD].shifts) { if (jobStats[s.jobName]) { jobStats[s.jobName].minutes -= s.actualWorkMin; jobStats[s.jobName].salary -= s.shiftSalary; } } dailyShifts[targetD] = { isWork: false, offReason: "法定休日", totalSalary: 0, shifts: [], intervals: dailyShifts[targetD].intervals, classCount: 0, eventCount: 0 }; break; } } consecutive = 0; for (let checkD = 1; checkD <= d; checkD++) { if (dailyShifts[checkD].isWork) consecutive++; else consecutive = 0; } } } else { consecutive = 0; } } let totalMonthSalary = 0; for (let d = 1; d <= daysInMonth; d++) { if (dailyShifts[d].isWork) totalMonthSalary += dailyShifts[d].totalSalary; } // HTML生成部分などはここにまとめることができます return { dailyShifts, jobStats, totalMonthSalary }; } function timeToMin(timeStr) { let parts = timeStr.split(":"); return parseInt(parts[0]) * 60 + parseInt(parts[1]); } function minToTime(min) { let h = Math.floor(min / 60); let m = min % 60; return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`; } function ceil15(min) { return Math.ceil(min / 15) * 15; } function floor15(min) { return Math.floor(min / 15) * 15; }
